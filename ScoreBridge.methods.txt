Initialize () {
AddToPluginsMenu('Codex Score Bridge', 'Run');
if (Sibelius.ProgramVersion >= 20200900) { SetInterpreterOption('TreatSingleCharacterAsString'); }
}

Count (items) {
countResult = 0;
for each item in items { countResult = countResult + 1; }
return countResult;
}

OpenScore (path) {
if (Sibelius.FileExists(path) = False) { return False; }
return Sibelius.Open(path, True);
}

ScoreStatus () {
score = Sibelius.ActiveScore;
if (IsValid(score) = False) { return CreateDictionary('active', False); }
return CreateDictionary('active', True, 'filename', '' & score.FileName, 'title', '' & score.Title, 'staves', 0 + score.StaffCount, 'bars', 0 + score.SystemStaff.BarCount, 'transposing', '' & score.TransposingScore);
}

ShowPart (filename, signature) {
score = Sibelius.ActiveScore;
if (IsValid(score) = False) { return False; }
if (score.FileName != filename) { return False; }
oldPart = score.CurrentDynamicPart;
part = FindPart(score, signature);
if (IsValid(part) = False) { score.CurrentDynamicPart = oldPart; return False; }
score.CurrentDynamicPart = part;
if (Sibelius.ShowDynamicPart(part, False) = False) { score.CurrentDynamicPart = oldPart; return False; }
return CreateDictionary('filename', '' & score.FileName, 'part', '' & score.PartName, 'signature', signature);
}

ProtocolTest () {
original = '%0A||' & Chr(10) & Chr(10) & Chr(13) & 'Bb ♭ %';
emptyFields = SplitString('left||right', '|', False); empty = Dec(emptyFields[1]); zeroRangeIterations = 0;
for z = 1 to 1 { zeroRangeIterations = zeroRangeIterations + 1; }
return CreateDictionary('version', '__VERSION__', 'build', '__BUILD__', 'original', original, 'encoded', Enc(original), 'roundtrip', Dec(Enc(original)), 'empty', empty, 'empty_equals', empty = '', 'zero_range_iterations', zeroRangeIterations);
}

GetLastResult () {
return Self._property:output;
}

RecoverJob (job) {
if ((Sibelius.FileExists(job & '/before.sib') = False) or (Sibelius.FileExists(job & '/request.tsv') = False)) { return False; }
request = Sibelius.ReadTextFile(job & '/request.tsv'); fields = SplitString(request[0], '|', False);
if ((Count(fields) != 14) or (fields[0] != 'CSB1') or ((fields[2] != 'APPLY') and (fields[2] != 'ENSURE_PART'))) { return False; }
original = Sibelius.ActiveScore;
if (IsValid(original) = False) { return False; }
if (original.FileName != Dec(fields[4])) { return False; }
Sibelius.CloseAllWindowsForScore(original, False);
return Sibelius.Open(job & '/before.sib', True);
}

Run () {
root = '__RUNTIME__/';
if (Sibelius.FileExists(root & 'pending.txt') = False) { return False; }
pointer = Sibelius.ReadTextFile(root & 'pending.txt');
if (Count(pointer) != 1) { return False; }
job = pointer[0];
if (Sibelius.FileExists(job & '/request.tsv') = False) { return False; }
if (Sibelius.FileExists(job & '/status.tsv')) { return False; }
Self._property:output = CreateDictionary('job', '' & job);
status = job & '/status.tsv';
Sibelius.CreateTextFile(status);
Sibelius.AppendLineToFile(status, 'RUNNING');
input = Sibelius.ReadTextFile(job & '/request.tsv');
if (Count(input) < 1) { return Fail(status, 'REFUSED', 'empty request'); }
rawHeader = SplitString(input[0], '|', False);
h = CreateSparseArray();
for each headerField in rawHeader { h.Push(Dec(headerField)); }
if (Count(h) != 14) { return Fail(status, 'REFUSED', 'invalid header'); }
Sibelius.AppendLineToFile(status, 'HEADER_READ');
if (h[0] != 'CSB1') { return Fail(status, 'REFUSED', 'protocol differs'); }
action = h[2];
if ((action != 'DISCOVER') and (action != 'INSPECT') and (action != 'EXPORT') and (action != 'EXPORT_SET') and (action != 'EXPORT_PROOF') and (action != 'PREVIEW') and (action != 'APPLY') and (action != 'ENSURE_PART')) { return Fail(status, 'REFUSED', 'unknown action'); }
score = Sibelius.ActiveScore;
if (IsValid(score) = False) { return Fail(status, 'REFUSED', 'no active score'); }
oldPart = score.CurrentDynamicPart;
full = null;
for each part in score.DynamicParts { if (part.IsFullScore) { full = part; } }
if (IsValid(full) = False) { return Fail(status, 'REFUSED', 'no full score'); }
score.CurrentDynamicPart = full;
if (action != 'DISCOVER') {
if ((score.Title != h[3]) or (score.FileName != h[4]) or (score.StaffCount != (0 + h[5])) or (score.SystemStaff.BarCount != (0 + h[6])) or (('' & score.TransposingScore) != h[7])) { return Finish(score, oldPart, status, 'REFUSED', 'active score contract differs'); }
}
partsPath = job & '/parts.tsv';
Sibelius.CreateTextFile(partsPath);
partIndex = 0;
for each part in score.DynamicParts {
score.CurrentDynamicPart = full;
members = CreateSparseArray();
for each Staff st in score { if (part.IncludesStaff(st)) { members.Push(st.StaffNum); } }
signature = members.Join(',');
score.CurrentDynamicPart = part;
Sibelius.AppendLineToFile(partsPath, partIndex & '|' & part.IsFullScore & '|' & Enc(score.PartName) & '|' & signature);
partIndex = partIndex + 1;
}
score.CurrentDynamicPart = full;
if (action = 'EXPORT_PROOF') {
identity = CreateSparseArray('SCORE|' & Enc(score.Title) & '|' & Enc(score.FileName) & '|' & score.StaffCount & '|' & score.SystemStaff.BarCount & '|' & score.TransposingScore);
WriteRows(job & '/identity.tsv', identity);
return ExportPartSet(score, oldPart, status, job, h[8], True);
}
before = Snapshot(score);
WriteRows(job & '/before.tsv', before);
if (action = 'DISCOVER') { return Finish(score, oldPart, status, 'DONE', 'discovery'); }
if (action = 'EXPORT_SET') { return ExportPartSet(score, oldPart, status, job, h[8], False); }
if (action = 'INSPECT') {
context = FindPart(score, h[8]);
if (IsValid(context) = False) { return Finish(score, oldPart, status, 'REFUSED', 'ambiguous or missing part'); }
score.CurrentDynamicPart = context;
WriteRows(job & '/context.tsv', Snapshot(score));
return Finish(score, oldPart, status, 'DONE', 'inspection');
}
if (action = 'EXPORT') {
context = FindPart(score, h[8]);
if (IsValid(context) = False) { return Finish(score, oldPart, status, 'REFUSED', 'ambiguous or missing part'); }
pdfPath = job & '/part.pdf';
if (Sibelius.FileExists(pdfPath)) { return Finish(score, oldPart, status, 'REFUSED', 'PDF already exists'); }
if (context.IsFullScore) { score.ExportScoreAsPDF(pdfPath); } else {
chosen = CreateSparseArray(); chosen.Push(context);
score.ExportPartsAsPDF(pdfPath, True, chosen, False);
}
score.CurrentDynamicPart = context;
WriteRows(job & '/context.tsv', Snapshot(score));
if (Sibelius.FileExists(pdfPath) = False) { return Finish(score, oldPart, status, 'FAILED', 'PDF absent'); }
return Finish(score, oldPart, status, 'EXPORTED_UNVERIFIED', 'host must open PDF and verify snapshot');
}
if (score.TransposingScore = False) { return Finish(score, oldPart, status, 'REFUSED', 'enable transposing score before editing'); }
if (Sibelius.FileExists(job & '/expected.tsv') = False) { return Finish(score, oldPart, status, 'REFUSED', 'no baseline'); }
expected = Sibelius.ReadTextFile(job & '/expected.tsv');
if (before.Join(Chr(10)) != JoinArray(expected, Chr(10))) { return Finish(score, oldPart, status, 'REFUSED', 'stale baseline: inspect again'); }
partEdit = ((action = 'PREVIEW') or (action = 'APPLY')) and (h[8] != 'score');
if (partEdit) {
context = FindPart(score, h[8]);
if (IsValid(context) = False) { return Finish(score, oldPart, status, 'REFUSED', 'ambiguous or missing edit part'); }
if (context.IsFullScore) { return Finish(score, oldPart, status, 'REFUSED', 'part edit requires dynamic part'); }
if (Sibelius.FileExists(job & '/expected-context.tsv') = False) { return Finish(score, oldPart, status, 'REFUSED', 'no part baseline'); }
score.CurrentDynamicPart = context;
contextBefore = Snapshot(score); WriteRows(job & '/context-before.tsv', contextBefore);
expectedContext = Sibelius.ReadTextFile(job & '/expected-context.tsv');
if (contextBefore.Join(Chr(10)) != JoinArray(expectedContext, Chr(10))) { return Finish(score, oldPart, status, 'REFUSED', 'stale part baseline: export again'); }
}
if (action = 'ENSURE_PART') {
members = SplitString(h[8], ',', False); previous = 0;
for each member in members { n = 0 + member; if ((('' & n) != member) or (n <= previous) or (n > score.StaffCount)) { return Finish(score, oldPart, status, 'REFUSED', 'invalid staff signature'); } previous = n; }
if ((Count(members) = 0) or (h[11] = '')) { return Finish(score, oldPart, status, 'REFUSED', 'part members/name required'); }
found = 0;
for each part in score.DynamicParts {
if (part.IsFullScore = False) {
signatureMembers = CreateSparseArray(); for each Staff st in score { if (part.IncludesStaff(st)) { signatureMembers.Push(st.StaffNum); } }
if (signatureMembers.Join(',') = h[8]) { found = found + 1; }
}
}
if (found > 1) { return Finish(score, oldPart, status, 'REFUSED', 'ambiguous existing parts'); }
if (found = 1) { return Finish(score, oldPart, status, 'DONE', 'matching part already exists'); }
backup = job & '/before.sib'; candidate = job & '/candidate.sib';
if (Sibelius.FileExists(backup) or Sibelius.FileExists(candidate)) { return Finish(score, oldPart, status, 'REFUSED', 'recovery/candidate exists'); }
score.SaveCopyAs(backup); if (Sibelius.FileExists(backup) = False) { return Finish(score, oldPart, status, 'REFUSED', 'backup absent'); }
Sibelius.AppendLineToFile(status, 'MUTATING');
createdPart = score.DynamicParts.CreatePartFromStaff(score.NthStaff(0 + members[0]));
if (IsValid(createdPart) = False) { return Finish(score, oldPart, status, 'FAILED_PARTIAL', 'part creation failed'); }
for i = 1 to Count(members) { if (createdPart.AddStaffToPart(score.NthStaff(0 + members[i])) = False) { return Finish(score, oldPart, status, 'FAILED_PARTIAL', 'part staff addition failed'); } }
score.CurrentDynamicPart = createdPart; score.PartName = h[11];
if ((score.PartName != h[11]) or (createdPart.StaveCount != Count(members))) { return Finish(score, oldPart, status, 'FAILED_PARTIAL', 'part identity differs'); }
score.CurrentDynamicPart = full;
Sibelius.AppendLineToFile(partsPath, partIndex & '|false|' & Enc(h[11]) & '|' & h[8]);
WriteRows(job & '/after.tsv', Snapshot(score));
score.SaveCopyAs(candidate);
if (Sibelius.FileExists(candidate) = False) { return Finish(score, oldPart, status, 'FAILED_PARTIAL', 'candidate absent'); }
return Finish(score, oldPart, status, 'APPLIED_UNVERIFIED', 'part created; host must verify unchanged full-score objects');
}
ops = CreateSparseArray();
keys = CreateSparseArray();
rhythmKeys = CreateSparseArray(); rhythmCount = 0; ordinaryCount = 0;
for i = 1 to Count(input) {
rawRow = SplitString(input[i], '|', False);
row = CreateSparseArray();
for each rowField in rawRow { row.Push(Dec(rowField)); }
if (Count(row) != 12) { return Finish(score, oldPart, status, 'REFUSED', 'invalid operation'); }

sn = 0 + row[2]; bn = 0 + row[3]; ordinal = 0 + row[4];
if ((sn < 0) or (sn > score.StaffCount) or (bn < 1) or (bn > score.SystemStaff.BarCount)) { return Finish(score, oldPart, status, 'REFUSED', 'target outside score'); }
if (row[6] = 'REPLACE_VOICE') {
rhythmCount = rhythmCount + 1; rhythmKey = sn & ':' & bn;
for each previous in rhythmKeys { if (previous = rhythmKey) { return Finish(score, oldPart, status, 'REFUSED', 'voice rebuilds require distinct staff/bar targets'); } }
rhythmKeys.Push(rhythmKey);
} else { ordinaryCount = ordinaryCount + 1; }
bar = GetBar(score, sn, bn);
target = null;
if (ordinal >= 0) {
count = 0;
for each item in bar { if (count = ordinal) { target = item; } count = count + 1; }
if (IsValid(target) = False) { return Finish(score, oldPart, status, 'REFUSED', 'target absent'); }
if (Fact(target) != row[5]) { return Finish(score, oldPart, status, 'REFUSED', 'target facts differ'); }
key = sn & ':' & bn & ':' & ordinal;
for each previous in keys { if (previous = key) { return Finish(score, oldPart, status, 'REFUSED', 'one operation per target per request'); } }
keys.Push(key);
}
if (partEdit) {
if (row[6] != 'SET') { return Finish(score, oldPart, status, 'REFUSED', 'part context allows graphical SET only'); }
k = row[7];
if ((k != 'Dx') and (k != 'Dy') and (k != 'RhDx') and (k != 'RhDy') and (k != 'Hidden') and (k != 'Magnetic')) { return Finish(score, oldPart, status, 'REFUSED', 'non-graphical part property'); }
}
if (Allowed(row, target, bar) = False) { return Finish(score, oldPart, status, 'REFUSED', 'unsupported operation or invalid value: ' & row[1]); }
record = CreateSparseArray(row.Slice(0), target); ops[ops.Length] = record.Slice(0);
}
if (ops.Length = 0) { return Finish(score, oldPart, status, 'REFUSED', 'no operations'); }
if ((rhythmCount > 0) and (ordinaryCount > 0)) { return Finish(score, oldPart, status, 'REFUSED', 'voice rebuilds cannot mix with other operations'); }
if (action = 'PREVIEW') { return Finish(score, oldPart, status, 'PREVIEWED', 'all native targets matched; no edits'); }
backup = job & '/before.sib'; candidate = job & '/candidate.sib';
if (Sibelius.FileExists(backup) or Sibelius.FileExists(candidate)) { return Finish(score, oldPart, status, 'REFUSED', 'recovery or candidate already exists'); }
score.SaveCopyAs(backup);
if (Sibelius.FileExists(backup) = False) { return Finish(score, oldPart, status, 'REFUSED', 'backup absent'); }
Sibelius.AppendLineToFile(status, 'MUTATING');
for i = 0 to ops.Length {
record = ops[i]; row = record[0]; target = record[1];
bar = GetBar(score, 0 + row[2], 0 + row[3]);
if (Apply(row, target, bar) = False) {
WriteRows(job & '/after.tsv', Snapshot(score));
return Finish(score, oldPart, status, 'FAILED_PARTIAL', 'do not save source; recover before.sib');
}
}
if (partEdit) { WriteRows(job & '/context-after.tsv', Snapshot(score)); }
score.CurrentDynamicPart = full;
WriteRows(job & '/after.tsv', Snapshot(score));
score.SaveCopyAs(candidate);
if (Sibelius.FileExists(candidate) = False) { return Finish(score, oldPart, status, 'FAILED_PARTIAL', 'candidate absent; recover before.sib'); }
return Finish(score, oldPart, status, 'APPLIED_UNVERIFIED', 'source remains unsaved; host must validate all observed changes');
}

ExportPartSet (score, oldPart, status, job, selectors, proofOnly) {
signatures = SplitString(selectors, ';', False);
if ((Count(signatures) < 1) or (Count(signatures) > 32)) { return Finish(score, oldPart, status, 'REFUSED', 'invalid export set size'); }
chosenParts = CreateSparseArray(); previousSignatures = CreateSparseArray();
for each signature in signatures {
for each previousSignature in previousSignatures { if (signature = previousSignature) { return Finish(score, oldPart, status, 'REFUSED', 'duplicate export signature'); } }
previousSignatures.Push(signature);
context = FindPart(score, signature);
if (IsValid(context) = False) { return Finish(score, oldPart, status, 'REFUSED', 'ambiguous or missing export part'); }
pdfPath = job & '/part-' & signature & '.pdf';
if (Sibelius.FileExists(pdfPath)) { return Finish(score, oldPart, status, 'REFUSED', 'export set PDF already exists'); }
chosenParts.Push(context);
}
exportIndex = 0;
for each signature in signatures {
context = chosenParts[exportIndex]; pdfPath = job & '/part-' & signature & '.pdf';
if (context.IsFullScore) { score.ExportScoreAsPDF(pdfPath); } else { chosen = CreateSparseArray(); chosen.Push(context); score.ExportPartsAsPDF(pdfPath, True, chosen, False); }
if (Sibelius.FileExists(pdfPath) = False) { return Finish(score, oldPart, status, 'FAILED', 'export set PDF absent'); }
exportIndex = exportIndex + 1;
}
if (proofOnly) { return Finish(score, oldPart, status, 'PDF_PROOF_EXPORTED', 'PDF proof only; musical state not inventoried or verified'); }
return Finish(score, oldPart, status, 'EXPORTED_UNVERIFIED', 'host must validate every PDF; shared full inventory, no per-part contexts');
}

Enc (value) {
s = '' & value;
s = utils.Replace(s, '%', '%25', True);
s = utils.Replace(s, '|', '%7C', True);
s = utils.Replace(s, Chr(13), '%0D', True);
s = utils.Replace(s, Chr(10), '%0A', True);
return s;
}

Dec (value) {
s = '' & value;
s = utils.Replace(s, '%0A', Chr(10), True);
s = utils.Replace(s, '%0D', Chr(13), True);
s = utils.Replace(s, '%7C', '|', True);
s = utils.Replace(s, '%25', '%', True);
return '' & s;
}

KV (key, value) {
return '|' & key & '=' & value;
}

Fact (obj) {
fact = 'Type=' & obj.Type & ('|Position=' & (obj.Position)) & ('|VoiceNumber=' & (obj.VoiceNumber)) & ('|Voices=' & (obj.Voices)) & ('|Hidden=' & (obj.Hidden)) & ('|CueSize=' & (obj.CueSize)) & ('|Dx=' & (obj.Dx)) & ('|Dy=' & (obj.Dy)) & ('|Magnetic=' & (obj.UsesMagneticLayout)) & ('|MagneticOverride=' & (obj.UsesMagneticLayoutSettingOverridden));
played = CreateSparseArray(); for pass = 1 to 10 { played.Push(obj.GetPlayedOnNthPass(pass)); }
fact = fact & ('|Played=' & (played.Join(',')));
if (obj.Type = 'NoteRest') {
activeInstrument = obj.ParentBar.GetInstrumentTypeAt(obj.Position);
fact = fact & ('|InstrumentStyle=' & (activeInstrument.StyleId));
fact = fact & ('|Duration=' & (obj.Duration));
fact = fact & ('|Grace=' & (obj.GraceNote));
fact = fact & ('|Slash=' & (obj.IsAcciaccatura));
fact = fact & ('|SingleTremolos=' & (obj.SingleTremolos));
fact = fact & ('|DoubleTremolos=' & (obj.DoubleTremolos));
fact = fact & ('|Beam=' & (obj.Beam));
fact = fact & ('|Stemweight=' & (obj.Stemweight)) & ('|StemFlipped=' & (obj.StemFlipped));
fact = fact & ('|RestPosition=' & (obj.RestPosition));
artics = CreateSparseArray(); for a = 0 to 16 { artics.Push(obj.GetArticulation(a)); }
fact = fact & ('|Articulations=' & (artics.Join(','))) & ('|NoteCount=' & (obj.NoteCount));
notes = CreateSparseArray();
for each note in obj {
noteFields = CreateSparseArray();
noteFields.Push(note.Pitch);
noteFields.Push(note.DiatonicPitch);
noteFields.Push(note.WrittenPitch);
noteFields.Push(note.WrittenDiatonicPitch);
noteFields.Push(note.Tied);
if (note.Tied) { noteFields.Push(note.TieStyle); } else { noteFields.Push('none'); }
noteFields.Push(note.NoteStyle);
noteFields.Push(note.AccidentalStyle);
noteFields.Push(note.Hidden);
noteFields.Push(note.Bracketed);
noteFields.Push('unavailable');
noteFields.Push('unavailable');
noteFields.Push('unavailable');
nf = noteFields.Join(',');
notes.Push(nf);
}
fact = fact & ('|Notes=' & (notes.Join(';')));
tuplet = obj.ParentTupletIfAny;
if (IsValid(tuplet)) { fact = fact & ('|Tuplet=' & (tuplet.Position & ',' & tuplet.Left & ',' & tuplet.Right & ',' & tuplet.Unit & ',' & obj.PositionInTuplet)); }
}
if (obj.Type = 'BarRest') { fact = fact & ('|RestType=' & (obj.RestType)); }
if ((obj.Type = 'Text') or (obj.Type = 'SystemTextItem')) { fact = fact & ('|StyleId=' & (obj.StyleId)) & ('|Text=' & (Enc(obj.Text))) & ('|Formatting=' & (Enc(obj.TextWithFormattingAsString))) & ('|EraseBackground=' & (obj.EraseBackground)); }
if (obj.IsALine) { fact = fact & ('|StyleId=' & (obj.StyleId)) & ('|Duration=' & (obj.Duration)) & ('|RhDx=' & (obj.RhDx)) & ('|RhDy=' & (obj.RhDy)) & ('|EndBar=' & (obj.EndBarNumber)) & ('|EndPosition=' & (obj.EndPosition)); }
if ((obj.Type = 'SymbolItem') or (obj.Type = 'SystemSymbolItem')) { fact = fact & ('|Index=' & (obj.Index)) & ('|Size=' & (obj.Size)); }
if (obj.Type = 'Tuplet') { fact = fact & ('|Left=' & (obj.Left)) & ('|Right=' & (obj.Right)) & ('|Unit=' & (obj.Unit)) & ('|PlayedDuration=' & (obj.PlayedDuration)) & ('|TupletStyle=' & (obj.Style)) & ('|Bracket=' & (obj.Bracket)) & ('|FullDuration=' & (obj.FullDuration)); }
if (obj.Type = 'Tuplet') { parentTuplet = obj.ParentTupletIfAny; if (IsValid(parentTuplet)) { fact = fact & ('|ParentTuplet=' & (parentTuplet.Position & ',' & parentTuplet.Left & ',' & parentTuplet.Right & ',' & parentTuplet.Unit & ',' & obj.PositionInTuplet)); } }
if (obj.Type = 'InstrumentChange') { fact = fact & ('|Instrument=' & (Enc(obj.FullInstrumentName))) & ('|TextLabel=' & (Enc(obj.TextLabel))); activeInstrument = obj.ParentBar.GetInstrumentTypeAt(obj.Position); fact = fact & ('|InstrumentStyle=' & (activeInstrument.StyleId)); }
if (obj.Type = 'Clef') { fact = fact & ('|StyleId=' & (obj.StyleId)); }
if (obj.Type = 'TimeSignature') { fact = fact & ('|Numerator=' & (obj.Numerator)) & ('|Denominator=' & (obj.Denominator)) & ('|TimeText=' & (Enc(obj.Text))); }
if (obj.Type = 'SpecialBarline') { fact = fact & ('|BarlineType=' & (obj.BarlineType)); }
if (obj.Type = 'KeySignature') { fact = fact & ('|Sharps=' & (obj.Sharps)) & ('|Major=' & (obj.Major)); }
return fact;
}

Snapshot (score) {
rows = CreateSparseArray();
Sibelius.CreateTextFile('__RUNTIME__/progress.tsv');
Sibelius.ResetStopWatch(41);
contextStaffCount = score.CurrentDynamicPart.StaveCount;
rows.Push('SCORE|' & Enc(score.Title) & '|' & Enc(score.FileName) & '|' & contextStaffCount & '|' & score.SystemStaff.BarCount & '|' & score.TransposingScore);
for sn = 0 to contextStaffCount + 1 {
Sibelius.AppendLineToFile('__RUNTIME__/progress.tsv', 'STAFF|' & sn & '|' & Sibelius.GetElapsedMilliSeconds(41));
if (sn = 0) { st = score.SystemStaff; } else { st = score.NthStaff(sn); rows.Push('STAFF|' & sn & '|' & Enc(st.InstrumentName) & '|' & Enc(st.InitialStyleId)); }
for bn = 1 to score.SystemStaff.BarCount + 1 {
bar = st.NthBar(bn);
if (sn = 0) { rows.Push('BAR|' & sn & '|' & bn & '|' & Enc(bar.ExternalBarNumberString) & '|' & bar.Length); } else {
it = bar.GetInstrumentTypeAt(0);
rows.Push('BAR|' & sn & '|' & bn & '|' & Enc(bar.ExternalBarNumberString) & '|' & bar.Length & '|' & bar.BreakType & '|' & bar.OnNthPage & '|' & bar.OnNthSystem & '|' & Enc(it.StyleId) & '|' & it.ChromaticTransposition & '|' & it.DiatonicTransposition);
}
ordinal = 0;
for each obj in bar { rows.Push('OBJ|' & sn & '|' & bn & '|' & ordinal & '|' & Enc(Fact(obj))); ordinal = ordinal + 1; }
}
}
Sibelius.AppendLineToFile('__RUNTIME__/progress.tsv', 'SNAPSHOT_COMPLETE|' & Sibelius.GetElapsedMilliSeconds(41));
return rows;
}

GetBar (score, sn, bn) {
if (sn = 0) { return score.SystemStaff.NthBar(bn); }
return score.NthStaff(sn).NthBar(bn);
}

FindPart (score, wanted) {
selector = SplitString(wanted, '#', False); signatureWanted = selector[0]; selectedIndex = -1;
if (Count(selector) = 2) { selectedIndex = 0 + selector[1]; }
full = null; for each part in score.DynamicParts { if (part.IsFullScore) { full = part; } }
score.CurrentDynamicPart = full;
match = null; matches = 0; partNumber = 0;
for each part in score.DynamicParts {
members = CreateSparseArray();
for each Staff st in score { if (part.IncludesStaff(st)) { members.Push(st.StaffNum); } }
sig = members.Join(',');
if (((signatureWanted = 'score') and part.IsFullScore) or ((signatureWanted = sig) and (part.IsFullScore = False) and ((selectedIndex = -1) or (selectedIndex = partNumber)))) { match = part; matches = matches + 1; }
partNumber = partNumber + 1;
}
if (matches != 1) { return null; }
return match;
}

WriteRows (path, rows) {
Sibelius.AppendLineToFile('__RUNTIME__/progress.tsv', 'RETURN_ROWS|' & path);
pathFields = SplitString(path, '/', False);
name = pathFields[Count(pathFields) - 1];
result = Self._property:output;
result[name] = rows;
Self._property:output = result;
Sibelius.AppendLineToFile('__RUNTIME__/progress.tsv', 'RETURN_ROWS_STORED');
}

Fail (status, code, message) {
Sibelius.AppendLineToFile(status, code & '|' & Enc(message)); return False;
}

Finish (score, oldPart, status, code, message) {
score.CurrentDynamicPart = oldPart;
